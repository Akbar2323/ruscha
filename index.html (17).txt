/* ------------------------------- Holat ----------------------------------- */

let pismoOpen   = null;   // ochilgan vazifa id
let pismoTimer  = null;
let pismoLeft   = 0;      // qolgan soniya
let pismoBusy   = false;
let pismoResult = null;

function pismoStop() {
  if (pismoTimer) { clearInterval(pismoTimer); pismoTimer = null; }
}

function pismoTask(id) { return PISMO_TASKS.find(t => t.id === id); }

function countWords(s) {
  return (s.trim().match(/[а-яёa-z0-9-]+/gi) || []).length;
}

/* ------------------------------ Ro'yxat ---------------------------------- */

function renderPismo() {
  const el = document.getElementById("view-pismo");

  if (pismoResult) { renderPismoResult(el); return; }
  if (pismoOpen)   { renderPismoEditor(el); return; }

  pismoStop();
  el.innerHTML = `
    <div class="panel">
      <h2>Письмо — ТРКИ-3 / C1</h2>
      <p class="subtitle">
        Uchta vazifa turi: norasmiy xat, qisqa axborot xabari va esse.
        Yozganingizni sun'iy intellekt ТРКИ mezonlari bo'yicha tekshiradi va
        xatolarni o'zbekcha tushuntiradi.
      </p>
      <p class="pismo-warn">
        Diqqat: bu baho taxminiy. U real ТРКИ ekspertining bahosi emas va
        sertifikat natijasini kafolatlamaydi — faqat tayyorgarlik uchun mo'ljallangan.
      </p>
    </div>

    <div class="cards">
      ${PISMO_TASKS.map(t => `
        <button class="card pismo-card" data-pismo="${t.id}">
          <span class="badge-kind">${escapeHtml(t.kind)}</span>
          <span class="pismo-card-title">${escapeHtml(t.title)}</span>
          <span class="badge-row">
            <span class="badge-time">${t.minutes} daqiqa</span>
            <span class="badge-level">${t.words[0]}–${t.words[1]} so'z</span>
          </span>
        </button>
      `).join("")}
    </div>
  `;

  el.querySelectorAll("[data-pismo]").forEach(b => {
    b.onclick = () => {
      pismoOpen = b.dataset.pismo;
      const t = pismoTask(pismoOpen);
      pismoLeft = t.minutes * 60;
      track("pismo-boshlandi-" + pismoOpen);
      renderPismo();
    };
  });
}

/* ------------------------------ Muharrir --------------------------------- */

function renderPismoEditor(el) {
  const t = pismoTask(pismoOpen);
  const saved = (state.pismoDrafts && state.pismoDrafts[t.id]) || "";

  el.innerHTML = `
    <div class="pismo-bar">
      <button class="btn btn-ghost btn-sm" id="pismoBack">← Orqaga</button>
      <span class="pismo-clock" id="pismoClock"></span>
    </div>

    ${t.source ? `
      <div class="panel pismo-source">
        <h3>Исходный текст</h3>
        <p>${escapeHtml(t.source)}</p>
      </div>` : ""}

    <div class="panel pismo-task">
      <h3>${escapeHtml(t.kind)}</h3>
      <pre class="pismo-task-text">${escapeHtml(t.task)}</pre>
    </div>

    <textarea id="pismoText" class="pismo-editor" spellcheck="false"
      placeholder="Начните писать здесь…">${escapeHtml(saved)}</textarea>

    <div class="pismo-foot">
      <span id="pismoCount" class="pismo-count"></span>
      <button class="btn btn-primary" id="pismoSend">Tekshirishga yuborish</button>
    </div>
  `;

  const ta    = el.querySelector("#pismoText");
  const cnt   = el.querySelector("#pismoCount");
  const clock = el.querySelector("#pismoClock");

  function updateCount() {
    const n = countWords(ta.value);
    const [lo, hi] = t.words;
    cnt.textContent = `${n} so'z (kerak: ${lo}–${hi})`;
    cnt.className = "pismo-count" + (n >= lo && n <= hi ? " ok" : n > hi ? " over" : "");
  }

  ta.addEventListener("input", () => {
    updateCount();
    state.pismoDrafts = state.pismoDrafts || {};
    state.pismoDrafts[t.id] = ta.value;
  });
  updateCount();

  function tick() {
    const m = Math.floor(pismoLeft / 60), s = pismoLeft % 60;
    clock.textContent = `${m}:${String(s).padStart(2, "0")}`;
    clock.classList.toggle("low", pismoLeft <= 120);
    if (pismoLeft <= 0) {
      pismoStop();
      toast("Vaqt tugadi — javobni yuborishingiz mumkin.");
      return;
    }
    pismoLeft--;
  }
  pismoStop();
  tick();
  pismoTimer = setInterval(tick, 1000);

  el.querySelector("#pismoBack").onclick = () => {
    pismoStop(); pismoOpen = null; saveState(); renderPismo();
  };

  el.querySelector("#pismoSend").onclick = () => sendPismo(t, ta.value);
}

/* ------------------------------ Yuborish --------------------------------- */

async function sendPismo(t, text) {
  if (pismoBusy) return;

  const n = countWords(text);
  if (n < t.words[0] * 0.6) {
    toast("Matn juda qisqa — avval vazifani to'liq bajaring.");
    return;
  }
  if (!currentUser) {
    toast("Tekshirish uchun hisobingizga kiring.");
    return;
  }

  pismoBusy = true;
  const btn = document.getElementById("pismoSend");
  btn.disabled = true;
  btn.textContent = "Tekshirilmoqda…";

  try {
    const token = await currentUser.getIdToken();
    const r = await fetch(PISMO_WORKER, {
      method: "POST",
      headers: { "Content-Type": "application/json", "Authorization": "Bearer " + token },
      body: JSON.stringify({ text, task: t.task, source: t.source })
    });
    const data = await r.json();

    if (!r.ok) { toast(data.message || "Xatolik yuz berdi."); return; }

    pismoStop();
    pismoResult = { task: t, data, text };
    state.pismoDone = state.pismoDone || [];
    if (!state.pismoDone.includes(t.id)) state.pismoDone.push(t.id);
    saveState();
    track("pismo-tekshirildi-" + t.id);
    renderPismo();
  } catch (e) {
    toast("Internet aloqasi yo'q yoki xizmat javob bermadi.");
  } finally {
    pismoBusy = false;
    if (btn) { btn.disabled = false; btn.textContent = "Tekshirishga yuborish"; }
  }
}

function renderPismoResult(el) {
  const { task, data } = pismoResult;
  const pct = Math.max(0, Math.min(100, data.total || 0));

  el.innerHTML = `
    <div class="panel pismo-score ${data.passed ? "pass" : "fail"}">
      <div class="pismo-total">${pct}<span>/100</span></div>
      <div class="pismo-verdict">${data.passed ? "Сдано" : "Не сдано"}</div>
      <div class="pismo-meta">
        Matnning haqiqiy darajasi: <b>${escapeHtml(data.cefr || "—")}</b> ·
        ${data.wordCount || 0} so'z ·
        o'tish chegarasi 66%
      </div>
      ${typeof data.remaining === "number"
        ? `<div class="pismo-left">Bugun yana ${data.remaining} ta tekshiruv qoldi</div>` : ""}
    </div>

    <div class="panel">
      <h3>Mezonlar bo'yicha</h3>
      ${Object.keys(PISMO_LABELS).map(k => {
        const s = (data.scores && data.scores[k]) || { score: 0, comment: "" };
        return `
          <div class="pismo-crit">
            <div class="pismo-crit-head">
              <span>${PISMO_LABELS[k]}</span>
              <b>${s.score}/25</b>
            </div>
            <div class="pismo-crit-bar"><i style="width:${(s.score / 25) * 100}%"></i></div>
            <p>${escapeHtml(s.comment || "")}</p>
          </div>`;
      }).join("")}
    </div>

    ${(data.errors && data.errors.length) ? `
      <div class="panel">
        <h3>Xatolar</h3>
        ${data.errors.map(e => `
          <div class="pismo-err ${e.severity === "high" ? "high" : "low"}">
            <div class="pismo-err-top">
              <span class="badge-kind">${escapeHtml(e.type || "")}</span>
              ${e.severity === "high"
                ? `<span class="pismo-err-flag">ma'noni buzadi</span>` : ""}
            </div>
            <div class="pismo-err-pair">
              <s>${escapeHtml(e.quote || "")}</s>
              <b>${escapeHtml(e.corrected || "")}</b>
            </div>
            <p>${escapeHtml(e.why || "")}</p>
          </div>`).join("")}
      </div>` : ""}

    ${(data.strengths && data.strengths.length) ? `
      <div class="panel">
        <h3>Kuchli tomonlar</h3>
        <ul class="pismo-list">${data.strengths.map(s => `<li>${escapeHtml(s)}</li>`).join("")}</ul>
      </div>` : ""}

    ${(data.nextSteps && data.nextSteps.length) ? `
      <div class="panel">
        <h3>Keyingi qadamlar</h3>
        <ul class="pismo-list">${data.nextSteps.map(s => `<li>${escapeHtml(s)}</li>`).join("")}</ul>
      </div>` : ""}

    ${data.improved ? `
      <div class="panel">
        <h3>Yaxshilangan variant</h3>
        <p class="pismo-improved">${escapeHtml(data.improved)}</p>
      </div>` : ""}

    <div class="pismo-foot">
      <button class="btn btn-ghost" id="pismoAgain">Qaytadan yozish</button>
      <button class="btn btn-primary" id="pismoList">Vazifalar ro'yxati</button>
    </div>
  `;

  el.querySelector("#pismoAgain").onclick = () => {
    pismoResult = null;
    pismoLeft = task.minutes * 60;
    renderPismo();
  };
  el.querySelector("#pismoList").onclick = () => {
    pismoResult = null; pismoOpen = null; renderPismo();
  };
}
